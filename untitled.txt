  <script>
    let userCoins = 1500;
    let loggedInUser = "Guest";

    // 1. Google Account Picker Trigger
    function triggerGoogleLogin() {
      google.accounts.id.prompt((notification) => {
        if (notification.isNotDisplayed() || notification.isSkippedMoment()) {
          google.accounts.id.renderButton(
            document.getElementById("auth-screen"),
            { theme: "filled_blue", size: "large" }
          );
        }
      });
    }

    // Decode Google Token
    function parseJwt(token) {
      const base64Url = token.split('.')[1];
      const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
      return JSON.parse(window.atob(base64));
    }

    // Handle Login Response
    function handleCredentialResponse(response) {
      const data = parseJwt(response.credential);
      loggedInUser = data.name;

      // Update UI
      document.getElementById('user-name').innerText = data.name;
      document.getElementById('user-avatar').src = data.picture;
      document.getElementById('profile-name-large').innerText = data.name;
      document.getElementById('profile-email-large').innerText = data.email;
      document.getElementById('profile-img-large').src = data.picture;

      // Switch to main app screen
      document.getElementById('auth-screen').style.display = 'none';
      document.getElementById('main-app').style.display = 'flex';
    }

    // Tab Navigation
    function switchTab(pageId, navEl) {
      document.querySelectorAll('.tab-page').forEach(page => page.classList.remove('active'));
      document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
      
      document.getElementById(pageId).classList.add('active');
      navEl.classList.add('active');
    }

    // Seat Toggle Logic
    function toggleSeat(seatEl, seatName) {
      if(seatEl.classList.contains('active')) {
        seatEl.classList.remove('active');
        seatEl.querySelector('.seat-icon').innerHTML = '<i class="fa-solid fa-plus"></i>';
        appendChat("System", `${loggedInUser} ${seatName} থেকে নেমে গেছেন।`);
      } else {
        seatEl.classList.add('active');
        seatEl.querySelector('.seat-icon').innerHTML = '<i class="fa-solid fa-microphone"></i>';
        appendChat("System", `${loggedInUser} ${seatName}-এ সিট দখল করেছেন!`);
      }
    }

    // Chat System
    function sendChatMessage() {
      const input = document.getElementById('chat-input');
      if(input.value.trim() !== "") {
        appendChat(loggedInUser, input.value);
        input.value = "";
      }
    }

    function appendChat(sender, text) {
      const chatBox = document.getElementById('chat-box');
      const msgDiv = document.createElement('div');
      msgDiv.className = 'chat-msg';
      msgDiv.innerHTML = `<span>${sender}:</span> ${text}`;
      chatBox.appendChild(msgDiv);
      chatBox.scrollTop = chatBox.scrollHeight;
    }

    // Gifting Logic
    function sendGift(giftName, cost) {
      if(userCoins >= cost) {
        userCoins -= cost;
        updateCoinsUI();
        appendChat("🎁 Gift", `${loggedInUser} ${giftName} পাঠিয়েছেন!`);
      } else {
        alert("আপনার পর্যাপ্ত কয়েন নেই! ওয়ালেট থেকে রিচার্জ করুন।");
      }
    }

    // Wallet Recharge Simulation
    function buyCoins(coins, price) {
      userCoins += coins;
      updateCoinsUI();
      alert(`সফলভাবে ${coins} কয়েন যুক্ত হয়েছে!`);
    }

    function updateCoinsUI() {
      document.getElementById('coin-balance').innerText = userCoins.toLocaleString();
      document.getElementById('wallet-coins').innerText = userCoins.toLocaleString();
    }

    // Leave Room & Logout
    function leaveRoom() {
      alert("আপনি লাইভ রুম থেকে লিভ নিয়েছেন।");
    }

    function logout() {
      document.getElementById('auth-screen').style.display = 'flex';
      document.getElementById('main-app').style.display = 'none';
    }
  </script>
